import json
import re

# ==========================================
# 1. THE TOOL (Action Capability)
# ==========================================
def get_stock_price(ticker: str) -> str:
    """Fetches the current real-time stock price for a given ticker symbol."""
    # A mock tool representing a real API call
    mock_market = {"AAPL": "$175.50", "GOOG": "$150.25", "MSFT": "$420.10"}
    price = mock_market.get(ticker.upper(), "Ticker not found")
    return f"The current price of {ticker.upper()} is {price}."

# Mapping tool names to their actual Python functions
AVAILABLE_TOOLS = {
    "get_stock_price": get_stock_price
}


# ==========================================
# 2. THE SKILL (Behavioral Framework)
# ==========================================
# This markdown text serves as a text-based "Skill" defining how the agent 
# must think, reason, and format its output (similar to ReAct prompting).
INVESTMENT_ANALYSIS_SKILL = """
Role: You are a professional financial analysis assistant.
Process: Follow this sequential process to answer the user query:
  1. THOUGHT: Figure out what information is missing.
  2. ACTION: If you need data, state the function to call in JSON format: {"tool": "tool_name", "arg": "value"}
  3. ANSWER: Once you have the data, provide a professional, concise summary.

Example Workflow:
Thought: I need Apple's stock price.
Action: {"tool": "get_stock_price", "arg": "AAPL"}
Observation: The current price of AAPL is $175.50.
Thought: I have the data to answer.
Answer: Apple Inc. is currently trading at $175.50.
"""


# ==========================================
# 3. MOCK LLM (Simulating Agent Brain)
# ==========================================
def mock_llm_call(prompt: str) -> str:
    """Simulates an LLM responding over multiple steps of a reasoning loop."""
    # Step 1: The agent notices it lacks data and requests a tool call
    if "Observation:" not in prompt:
        return 'Thought: I need the stock price of Apple to proceed.\nAction: {"tool": "get_stock_price", "arg": "AAPL"}'
    
    # Step 2: The agent processes the tool's result and delivers the final answer
    if "The current price of AAPL is $175.50." in prompt:
        return 'Thought: I have the current price. I can now finalize the analysis.\nAnswer: Apple Inc. (AAPL) is performing steadily, currently trading at $175.50.'
    
    return "Answer: I encountered an unexpected processing state."


# ==========================================
# 4. THE AGENT REASONING LOOP (Execution)
# ==========================================
def run_agent_loop(user_query: str):
    print(f"🚀 User Query: '{user_query}'\n")
    
    # Initialize the short-term working context with the specialized skill
    agent_context = f"{INVESTMENT_ANALYSIS_SKILL}\nUser Query: {user_query}\n"
    
    # Run the continuous reasoning loop (Plan -> Act -> Observe -> Revise)
    for step in range(1, 4):
        print(f"--- [Loop Iteration {step}] ---")
        
        # Brain: Ask the LLM what to do next based on the skill and context
        llm_response = mock_llm_call(agent_context)
        print(llm_response)
        
        # Append the LLM's thought process to history
        agent_context += f"\n{llm_response}"
        
        # Check if the LLM arrived at the final answer
        if "Answer:" in llm_response:
            print("\n✅ Task completed successfully!")
            break
            
        # Parse and execute action if a tool call is requested
        action_match = re.search(r"Action:\s*(\{.*\})", llm_response)
        if action_match:
            try:
                tool_call = json.loads(action_match.group(1))
                tool_name = tool_call["tool"]
                tool_arg = tool_call["arg"]
                
                if tool_name in AVAILABLE_TOOLS:
                    # Execute tool
                    observation = AVAILABLE_TOOLS[tool_name](tool_arg)
                    print(f"🔍 [Tool Output/Observation]: {observation}")
                    
                    # Feed observation back into context
                    agent_context += f"\nObservation: {observation}"
                else:
                    agent_context += f"\nObservation: Tool '{tool_name}' is not available."
            except Exception as e:
                agent_context += f"\nObservation: Failed to parse action syntax. Error: {str(e)}"

# Execute the agent
run_agent_loop("Can you check the current market valuation for Apple?")
